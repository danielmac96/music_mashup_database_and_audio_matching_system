# Mashup Engine — Python dependencies
#
# Python 3.10+ required (yt-dlp drops older versions over time).
# External binary: ffmpeg + ffprobe must be on PATH (audio extraction + duration
# checks). Install via your OS package manager, e.g. `brew install ffmpeg` or
# `apt install ffmpeg`. The web app's Import tab shows a warning if any of these
# are missing — see the "First run" section in readme.md.

# ── Download + ingest ─────────────────────────────────────────────────────────
yt-dlp>=2026.7.4           # SoundCloud/YouTube metadata + audio. SoundCloud
                           # changes its API often — stale yt-dlp breaks downloads
                           # (the Import tab warns and offers one-click update).

# ── Stem separation (Demucs + PyTorch) ────────────────────────────────────────
torch==2.5.1
torchaudio==2.5.1
demucs==4.0.1              # htdemucs, four sources; ~400MB weights on first run
diffq-fixed>=0.2 ; sys_platform == "win32"   # a Demucs dependency
diffq>=0.2 ; sys_platform != "win32"

# ── Audio analysis ────────────────────────────────────────────────────────────
numpy==1.26.4              # pinned <2 for librosa
scipy==1.12.0
soundfile==0.12.1
librosa==0.10.1            # helpers: decode, renders, the fallback section cut
setuptools==80.10.2        # librosa 0.10.1 imports pkg_resources, which newer
                           # setuptools (84 in a clean CI install) no longer ships
audioread>=3.0.0           # MP3 decode fallback for librosa
tqdm==4.66.4
mutagen>=1.47              # ID3 TBPM/TKEY on exported FL session stems, so the
                           # folder is also usable from Serato/rekordbox.
                           # Optional at runtime: a missing mutagen skips tagging
                           # rather than failing the export.

# ── Learned pairwise scorer (Phase 5) ─────────────────────────────────────────
scikit-learn>=1.3,<2     # HistGradientBoostingClassifier + LogisticRegression baseline
threadpoolctl>=3.1          # analysis/threads.py: per-worker BLAS/OpenMP cap (also a scikit-learn dependency)
joblib>=1.3.0            # model artifact (de)serialisation
